cmake_minimum_required(VERSION 3.20)
project(orqest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

add_library(orqest src/orqest.cpp src/wire.cpp)
target_include_directories(orqest PUBLIC include)
target_compile_options(orqest PRIVATE -Wall -Wextra -Wpedantic)

add_executable(orqest_tests tests/orqest_tests.cpp)
target_link_libraries(orqest_tests PRIVATE orqest Threads::Threads)
add_executable(orqest_harness_tests tests/orqest_harness_tests.cpp)
target_link_libraries(orqest_harness_tests PRIVATE orqest Threads::Threads)

add_executable(orqest-ric apps/orqest_ric.cpp)
target_link_libraries(orqest-ric PRIVATE orqest Threads::Threads sctp)
add_executable(orqest-du apps/orqest_du.cpp)
target_link_libraries(orqest-du PRIVATE orqest Threads::Threads sctp)
foreach(target orqest-ric orqest-du)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_link_options(${target} PRIVATE -Wl,-z,noexecstack)
endforeach()

enable_testing()
add_test(NAME orqest_tests COMMAND orqest_tests)
add_test(NAME orqest_harness_tests COMMAND orqest_harness_tests)
