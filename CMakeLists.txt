cmake_minimum_required(VERSION 3.20)
project(orqest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(orqest src/orqest.cpp)
target_include_directories(orqest PUBLIC include)
add_executable(orqest_tests tests/orqest_tests.cpp)
target_link_libraries(orqest_tests PRIVATE orqest)
enable_testing()
add_test(NAME orqest_tests COMMAND orqest_tests)
